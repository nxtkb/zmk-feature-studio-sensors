if(CONFIG_NXTKB_STUDIO_SENSORS)
  file(STRINGS "${APPLICATION_SOURCE_DIR}/include/zmk/keymap.h" SS_API_MARKER REGEX "define ZMK_STUDIO_SENSORS_LOCAL_API 1")
  if(NOT SS_API_MARKER)
    message(FATAL_ERROR "Apply the local Studio sensors integration patch to the pinned official ZMK checkout first")
  endif()
  file(SHA256 "${PROJECT_BINARY_DIR}/zephyr.dts" SS_DTS_HASH)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/protocol/local/sensors.proto" SS_SCHEMA_HASH)
  file(SHA256 "${PROJECT_BINARY_DIR}/include/generated/zephyr/autoconf.h" SS_CONFIG_HASH)
  set(SS_SOURCE_HASHES "")
  file(GLOB_RECURSE SS_BEHAVIOR_FILES CONFIGURE_DEPENDS "${APPLICATION_SOURCE_DIR}/src/behaviors/*.c" "${APPLICATION_SOURCE_DIR}/src/behaviors/*.h")
  foreach(module_dir IN LISTS ZMK_EXTRA_MODULES)
    file(GLOB_RECURSE SS_MODULE_FILES CONFIGURE_DEPENDS "${module_dir}/src/*.c" "${module_dir}/include/*.h")
    list(APPEND SS_BEHAVIOR_FILES ${SS_MODULE_FILES})
  endforeach()
  set(SS_CANONICAL_FILES "")
  foreach(source IN LISTS SS_BEHAVIOR_FILES)
    file(REAL_PATH "${source}" source_real)
    list(APPEND SS_CANONICAL_FILES "${source_real}")
  endforeach()
  set(SS_BEHAVIOR_FILES ${SS_CANONICAL_FILES})
  list(REMOVE_DUPLICATES SS_BEHAVIOR_FILES)
  list(SORT SS_BEHAVIOR_FILES)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    ${SS_BEHAVIOR_FILES} "${CMAKE_CURRENT_SOURCE_DIR}/protocol/local/sensors.proto")
  foreach(source IN LISTS SS_BEHAVIOR_FILES)
    file(SHA256 "${source}" hash)
    string(APPEND SS_SOURCE_HASHES "${hash}")
  endforeach()
  string(SHA256 SS_STORAGE_HASH "${SS_DTS_HASH}:${SS_SCHEMA_HASH}:${SS_CONFIG_HASH}:${SS_SOURCE_HASHES}:storage-v1")
  set(SS_SIGNATURE "")
  foreach(index RANGE 0 15)
    math(EXPR offset "${index} * 2")
    string(SUBSTRING "${SS_STORAGE_HASH}" ${offset} 2 octet)
    string(APPEND SS_SIGNATURE "0x${octet},")
  endforeach()
  configure_file(include/studio_sensors/build_config.h.in ${CMAKE_CURRENT_BINARY_DIR}/ss_build_config.h @ONLY)
  target_include_directories(app PRIVATE include ${CMAKE_CURRENT_BINARY_DIR})
  target_compile_definitions(app PRIVATE SS_MAX_SENSORS=${CONFIG_NXTKB_STUDIO_SENSORS_MAX_SENSORS})
  target_sources(app PRIVATE src/engine.c src/rpc.c src/zephyr.c)
endif()
